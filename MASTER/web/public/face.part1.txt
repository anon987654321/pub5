function nearestRateIndex(rate) {
  const value = Number.isFinite(rate) && rate > 0 ? rate : 1.25;
  let best = 0;
  let bestDelta = Infinity;
  for (let i = 0; i < RATES.length; i++) {
    const delta = Math.abs(RATES[i] - value);
    if (delta < bestDelta) { best = i; bestDelta = delta; }
  }
  return best;
}

const SENT_BREAK = /([.!?…]+["'\u201D]?\s+|[\n]{2,})/;
const TTS_MIN_CHUNK = 48;
function pullStreamingTtsChunk(pending) {
  if (!pending) return { chunk: '', rest: '' };
  const sentBreak = pending.match(SENT_BREAK);
  if (!sentBreak) return { chunk: '', rest: pending };
  const cut = sentBreak.index + sentBreak[0].length;
  return { chunk: pending.slice(0, cut).trim(), rest: pending.slice(cut) };
}
function shouldEnqueueTtsChunk(text, opts = {}) {
  const t = String(text || '').trim();
  if (!t) return false;
  if (opts.flush) return true;
  if (t.length >= TTS_MIN_CHUNK) return true;
  if (/[.!?…]["'\u201D]?$/.test(t)) return true;
  return false;
}
function looksLikeListingStream(text) {
  const t = String(text || '');
  if (looksLikeProse(t)) return false;
  const lines = t.split('\n').map((l) => l.trim()).filter(Boolean);
  const tokens = t.split(/\s+/).filter(Boolean);
  if (tokens.length >= 8) {
    const short = tokens.filter((tok) => tok.length <= 28 && !/[.!?…]/.test(tok)).length;
    const times = tokens.filter((tok) => /^\d{1,2}:\d{2}$/.test(tok)).length;
    const paths = tokens.filter((tok) => /[./\\]/.test(tok)).length;
    const nums = tokens.filter((tok) => /^\d+$/.test(tok)).length;
    if (short / tokens.length >= 0.85 && times >= 3 && (paths >= 3 || nums / tokens.length >= 0.4)) return true;
  }
  if (lines.length < 3) return false;
  const tabular = lines.filter((l) => /^\d+\s+\d+\s+/.test(l) || /^[\d.:+\-]+\s+[\w./-]+$/.test(l)).length;
  return tabular / lines.length >= 0.6;
}
function looksLikeProse(text) {
  const t = String(text || '').trim();
  if (t.length < 64) return false;
  const sentences = (t.match(/[^.!?…]+[.!?…]+/g) || []).length;
  return sentences >= 2 || /\b(the|this|that|you|we|I|here|there|because|however)\b/i.test(t);
}
function shouldSpeakStreamReply(text, ttsSuppressed) {
  const t = String(text || '').trim();
  if (!t) return false;
  if (!ttsSuppressed) return true;
  return looksLikeProse(t);
}
// A reasoning model appends its working as <think>...</think> after the answer
// (ruby_llm_sender.rb does this for Nemotron 3). Read aloud, it turned a
// one-line reply into a paragraph of "We need to answer..." spoken after it.
// Cutting a closed block and an unclosed tail keeps the text prefix-stable
// while the stream grows, so the spoken-length cursor below stays valid.
function withoutThinking(text) {
  return String(text || '').replace(/<think>[\s\S]*?(?:<\/think>|$)/gi, '');
}
function pushLiveStreamTts(pending, spokenLen) {
  if (!ttsStreamLiveEnabled?.()) return spokenLen;
  const slice = withoutThinking(pending).slice(spokenLen);
  if (!slice || looksLikeListingStream(pending)) return spokenLen;
  const m = slice.match(/^[\s\S]*?[.!?…]["'\u201D]?\s+/);
  if (!m) return spokenLen;
  const chunk = m[0].trim();
  // Sentence completion is the safety boundary; short answers should not wait
  // for the whole model response merely to clear an arbitrary character count.
  enqueueSpeech(chunk, { flush: true });
  return spokenLen + m[0].length;
}
function flushStreamTts(pending, opts = {}) {
  const text = withoutThinking(pending);
  if (!text.trim() || looksLikeListingStream(text)) return;
  const alreadySpoken = Math.max(0, Math.min(text.length, Number(opts.spokenLen) || 0));
  let rest = text.slice(alreadySpoken).trimStart();
  let first = true;
  let pulled;
  const tsPrefix = opts.prependTimestamp
    ? `As of ${new Date().toLocaleDateString('en-GB', { day: 'numeric', month: 'long', year: 'numeric' })}. `
    : '';
  const speakOpts = { flush: true };
  while ((pulled = pullStreamingTtsChunk(rest)).chunk) {
    rest = pulled.rest;
    const prefix = first && alreadySpoken === 0 ? tsPrefix : '';
    first = false;
    enqueueSpeech(prefix + pulled.chunk, speakOpts);
  }
  if (rest.trim()) enqueueSpeech((first ? tsPrefix : '') + rest.trim(), speakOpts);
}

}

let renderer, scene, camera;
{
  try {
    // antialias: false, deliberately. Nothing here has an edge to smooth — the
    // scene is 1px points and one full-screen quad. Worse, MSAA is what stopped
    // a 1px point being one pixel: a point whose centre lands between pixels
    // resolves as partial coverage across up to four of them, so every particle
    // came out dim and smeared rather than crisp. Multisampling a pixel-art
    // buffer costs bandwidth to destroy the thing being drawn.
    renderer = new THREE.WebGLRenderer({ canvas: cv, antialias: false, alpha: false, depth: true, stencil: false, preserveDrawingBuffer: false, powerPreference: 'high-performance' });
    renderer.setClearColor(0x000000, 1);
    renderer.autoClear = true;
  } catch (e) {
    window.__MASTER_RENDERER_FAILED__ = true;
    window.MASTER?.boot?.signal?.('renderer_failed', { source: 'WebGLRenderer' });
    throw e;
  }
  scene = new THREE.Scene();
  camera = new THREE.PerspectiveCamera(FACE_CAMERA_FOV, 1, 0.1, 100);
  camera.position.set(0, 0, FACE_CAMERA_DISTANCE);
}
rootBody.dataset.faceRenderer = 'webgl';

let W = 0, H = 0, DPR = 1;
let faceBufferKey = '';
function resize(force = false) {
  if (!renderer) return;
  const width = cv.clientWidth, height = cv.clientHeight;
  if (width <= 0 || height <= 0) return;
  W = width; H = height;
  DPR = faceRenderDpr();

  const profile = rootBody.dataset.runtimeProfile || 'full';
  FACE_RENDER_SCALE = Math.max(FACE_RENDER_SCALE_MIN, Math.min(FACE_RENDER_SCALE_MAX, FACE_RENDER_SCALE));
  const conserving = faceLowBattery || profile === 'battery';
  const requested = conserving ? Math.min(FACE_RENDER_SCALE, 0.48) : FACE_RENDER_SCALE;
  const scale = Math.min(conserving ? requested : Math.max(requested, 280 / W, 200 / H),
    FACE_BUFFER_MAX_W / W, FACE_BUFFER_MAX_H / H);
  const internalW = Math.max(1, Math.round(W * scale));
  const internalH = Math.max(1, Math.round(H * scale));
  const key = [internalW, internalH, DPR, W / H].join(':');
  if (force !== true && key === faceBufferKey) return;
  renderer.setPixelRatio(DPR);
  renderer.setSize(internalW, internalH, false);
  camera.aspect = W / H;
  camera.updateProjectionMatrix();
  faceBufferKey = key;

  // A const cv here shadowed the canvas binding for the whole function, so the
  // first line of resize() threw "Cannot access cv before initialization".
  const canvas = renderer.domElement;
  canvas.style.imageRendering = "pixelated";
}
window.addEventListener('resize', resize, { passive: true });
if ('ResizeObserver' in window && cv) {
  new ResizeObserver(() => resize()).observe(cv.parentElement || cv);
}

// Grayscale depth map: white = near (high Z), black = background (filtered)
// Phantom.land technique: sample pixel luminance directly as Z, no edge detection
// Speculative distant-human phenotype: a restrained extension of documented Homo trends —
// fuller neurocranial vault, more retracted lower face, reduced brow/nasal projection,
// and a slightly stronger orbital field for visually mediated communication.
// This is a design hypothesis, not a biological prediction.
// Sample the selected solid directly, including its rear surfaces and cavities.
function sculptParticleVolume(count) {
  const home = new Float32Array(count * 3), scatter = new Float32Array(count * 3);
  const seeds = new Float32Array(count), curvature = new Float32Array(count);
  const boundary = new Float32Array(count), zone = new Float32Array(count);
  const masks = window.MASTER_FACE_MASKS;
  const profile = masks?.current()?.id;
  if (!profile) throw new Error('particle mask catalogue not initialized');
  for (let i = 0; i < count; i++) {
    const point = masks.sample(profile, i);
    if (!point || point.length !== 3 || !point.every(Number.isFinite)) throw new Error('invalid mask particle');
    const [x,y,z] = point;
    const seed = (Math.imul(i + 1, 2654435761) >>> 0) / 4294967296;
    home.set([x,y,z], i * 3);
    scatter.set([x,y,z], i * 3);
    seeds[i] = seed * Math.PI * 2;
    curvature[i] = seed;
    boundary[i] = Math.abs(z) > .22 ? 1 : 0;
    zone[i] = y < -.14 ? .8 : y > .1 && Math.abs(x) > .09 ? .5 : .25;
  }
  return {home, scatter, seeds, curvature, boundary, zone};
}

function particleScale() {
  const area = Math.max(320 * 480, (cv.clientWidth || window.innerWidth) * (cv.clientHeight || window.innerHeight));
  const dpr = Math.min(window.devicePixelRatio || 1, 1.25);
  const areaScale = Math.sqrt((area * dpr) / (1280 * 720));
  const budget = Number(faceRenderBudget().particles || 200);
  const budgetScale = Math.sqrt(Math.max(0.25, budget / 200));
  return Math.max(0.7, Math.min(1.35, areaScale, budgetScale));
}
// The primary field replaces the stipple population, rather than adding to it.
const primaryCeiling = Math.max(1, Math.floor(Number(faceRenderBudget().stipple_points) ||
  (State.coarsePointer ? 2400 : 8100)));
let faceHome, faceScatter, faceSeeds, faceCurvature, faceBoundary, faceZone;
({ home: faceHome, scatter: faceScatter, seeds: faceSeeds,
   curvature: faceCurvature, boundary: faceBoundary, zone: faceZone } =
  sculptParticleVolume(Math.min(primaryCeiling, State.coarsePointer ? 2400 : 6400)));
